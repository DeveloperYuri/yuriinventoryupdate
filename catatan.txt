// Buat view item table spare parts, asset, atk (Budi)

CREATE OR REPLACE VIEW v_items_all AS

-- SPARE PARTS (PAKAI STOK)
SELECT
    sp.id,
    sp.name,
    'sparepart' AS item_type,
    sp.stock
FROM spare_parts sp

UNION ALL

-- ASSET TOOLS (TANPA STOK)
SELECT
    at.id,
    at.name,
    'asset' AS item_type,
    0 AS stock
FROM asset_tools at

UNION ALL

-- ATK (PAKAI STOK)
SELECT
    a.id,
    a.name,
    'atk' AS item_type,
    a.stock
FROM atk a;

// buat count jumlah jenis spare part baru per bulan (Budi) 

SELECT 
    ROW_NUMBER() OVER (ORDER BY created_at ASC) AS no,
    id,
    name,
    stock,
    created_at
FROM spare_parts
WHERE MONTH(created_at) = 12
  AND YEAR(created_at) = 2025;

// Hitung total Masuk Per Bulan (Budi)

SELECT SUM(quantity) AS total_masuk
FROM stock_transactions
WHERE type = 'masuk'
  AND MONTH(created_at) = 1
  AND YEAR(created_at) = 2026;

//Hitung top 5 spare part yang keluar setiap bulan (Budi)
  SELECT 
    spare_part_id,
    SUM(quantity) AS total
FROM stock_transactions
WHERE LOWER(type) = 'out'
  AND MONTH(created_at) = 12
  AND YEAR(created_at) = 2025
GROUP BY spare_part_id
HAVING total > 0
ORDER BY total DESC
LIMIT 5;

@push('scripts')
    <script src="https://cdn.jsdelivr.net/npm/litepicker/dist/bundle.js"></script>
    <script>
        let itemIndex = 0;

        // 1. Litepicker Initialization
        new Litepicker({
            element: document.getElementById('tanggalMulai'),
            lang: 'id',
            format: 'DD MMMM YYYY',
            dropdowns: {
                minYear: 2020,
                maxYear: new Date().getFullYear() + 5,
                months: true,
                years: true
            },
            setup: picker => picker.on('selected', date => {
                document.getElementById('tanggalHidden').value = date.format('YYYY-MM-DD');
            })
        });

        // 2. Build Row Function (Disesuaikan dengan field database)
        function buildRow({
            sparepart_id = '',
            nama = '',
            qty = 1,
            maxQty = null,
            source = 'manual'
        }) {
            const index = itemIndex++;
            return `
    <tr class="data-row">
        <td>
            <input type="hidden" name="items[${index}][sparepart_id]" value="${sparepart_id}">
            <input type="hidden" name="items[${index}][source]" value="${source}">
            <input type="text" name="items[${index}][nama_real]" class="form-control" value="${nama}" ${source === 'sp' ? '' : ''}>
        </td>
        <td>
            <input type="number" name="items[${index}][qty_terima]" class="form-control" min="1" ${maxQty ? `max="${maxQty}"` : ''} value="${qty}">
        </td>
        <td>
            <button type="button" class="btn btn-danger btn-sm removeLine">Remove</button>
        </td>
    </tr>`;
        }

        // 3. Add Manual Spare Part
        $('#addLineBtn').on('click', function(e) {
            e.preventDefault();
            const $row = $(buildRow({
                source: 'manual'
            }));
            $('#productTable tbody').find('tr:last').before($row);

            // Terapkan autocomplete hanya pada input teks di baris baru ini
            applyAutocomplete($row.find('input[name$="[nama_real]"]'));

            $row.find('.removeLine').on('click', function() {
                $row.remove();
            });
        });

        // 4. Autocomplete Manual Input
        function applyAutocomplete(el) {
            el.autocomplete({
                source: function(request, response) {
                    $.getJSON('/spareparts/search', {
                        q: request.term
                    }, response);
                },
                minLength: 1,
                select: function(event, ui) {
                    $(this).val(ui.item.label);
                    $(this).siblings('input[name$="[sparepart_id]"]').val(ui.item.id);
                    return false;
                }
            });
        }

        // 5. SP Autocomplete (Generate Rows Otomatis)
        $('#po_autocomplete').autocomplete({
            source: function(request, response) {
                $.getJSON("{{ route('suratpesanan.search') }}", {
                    q: request.term
                }, response);
            },
            minLength: 1,
            select: function(event, ui) {
                $('#po_autocomplete').val(ui.item.label);
                $('#po_id').val(ui.item.id);

                // Loader sederhana
                const tbody = $('#productTable tbody');
                tbody.find('tr.data-row').remove();

                $.getJSON(`/surat-pesanan/${ui.item.id}/items`, function(items) {
                    if (items.length === 0) {
                        alert("Tidak ada item sisa pada SP ini.");
                        return;
                    }

                    items.forEach(item => {
                        const $row = $(buildRow({
                            sparepart_id: item
                            .sparepart_id, // Dari mapping controller: spare_part_id
                            nama: item.nama, // Dari mapping controller: nama_item
                            qty: item
                            .qty_sisa, // Dari mapping controller: qty_kurang
                            maxQty: item.qty_sisa,
                            source: 'sp'
                        }));

                        tbody.find('tr:last').before($row);
                        $row.find('.removeLine').on('click', function() {
                            $row.remove();
                        });
                    });
                }).fail(function() {
                    alert("Gagal mengambil data item SP. Pastikan koneksi/route benar.");
                });

                return false;
            }
        });

        // 6. Submit Button Loading
        $('#myForm').on('submit', function() {
            $('#saveBtn').prop('disabled', true);
            $('#btnText').html('<span class="spinner-border spinner-border-sm"></span> Saving...');
        });

        // 7. Prevent Enter Submit
        $("#myForm").on("keydown", function(event) {
            if (event.key === "Enter" && event.target.tagName !== "TEXTAREA") {
                event.preventDefault();
            }
        });
    </script>
